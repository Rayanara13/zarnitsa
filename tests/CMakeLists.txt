find_package(Qt5 REQUIRED COMPONENTS Test)

function(zr_test name)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE zarnitsa_core Qt5::Test)
  target_compile_definitions(${name} PRIVATE ZR_HELPER_PATH="$<TARGET_FILE:zarnitsa-icmp>" ZR_TESTDATA="${CMAKE_CURRENT_SOURCE_DIR}/data")
  add_dependencies(${name} zarnitsa-icmp)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;LC_ALL=C.UTF-8" TIMEOUT 120)
endfunction()

zr_test(test_core)
zr_test(test_integrations)
